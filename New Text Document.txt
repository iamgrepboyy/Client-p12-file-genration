def detect_key_format(key_path):
    data = key_path.read_bytes()

    pem_markers = [
        b"-----BEGIN PRIVATE KEY-----",
        b"-----BEGIN ENCRYPTED PRIVATE KEY-----",
        b"-----BEGIN RSA PRIVATE KEY-----",
        b"-----BEGIN EC PRIVATE KEY-----",
        b"-----BEGIN DSA PRIVATE KEY-----",
    ]

    if any(x in data for x in pem_markers):
        return "PEM"

    for inform in ("DER",):
        p = run(["openssl", "pkey", "-inform", inform, "-in", str(key_path), "-noout"])

def sha256_public_key_from_cert(cert_path, inform=None):
    cmd = ["openssl", "x509"]
    if inform:
        cmd += ["-inform", inform]
    cmd += ["-in", str(cert_path), "-pubkey", "-noout"]
    p = run(cmd)
    if p.returncode != 0:
        return None, p.stderr.strip()

    pub = p.stdout.encode()
    # Normalize by hashing the DER SubjectPublicKeyInfo.
    with tempfile.NamedTemporaryFile(mode="wb", suffix=".pub.pem", delete=False) as f:
        f.write(pub)
        pubfile = f.name

    try:
        q = run(["openssl", "pkey", "-pubin", "-in", pubfile, "-outform", "DER"])
        if q.returncode != 0:
            return None, q.stderr.strip()
        return hashlib.sha256(q.stdout).hexdigest(), None
    finally:
        os.unlink(pubfile)

    cmd += ["-in", str(key_path), "-out", str(out_path)]

    if key_password is not None:
        cmd += ["-passin", "stdin"]
        result = run(cmd, stdin_text=key_password + "\n")
    else:
        result = run(cmd)

    if result.returncode != 0:
        fail("Could not parse/normalize the private key:\n" + result.stderr.strip())


def validate_private_key(
    key_path: Path, key_format: str
) -> tuple[bool, str | None]:
    cmd = ["openssl", "pkey"]
    if key_format == "DER":
        cmd += ["-inform", "DER"]
    cmd += ["-in", str(key_path), "-noout"]

    result = run(cmd)
    if result.returncode == 0:
        return True, None

    error = (result.stderr or "").lower()

    encrypted = (
        b"ENCRYPTED PRIVATE KEY" in key_path.read_bytes()
        or "bad decrypt" in error
        or "bad password" in error
        or "maybe wrong password" in error
        or "pass phrase" in error
        or "password" in error
    )

    return False, "encrypted" if encrypted else result.stderr.strip()


def public_key_hash_from_cert(cert_pem: Path) -> str:
    result = run([
        "openssl", "x509",
        "-in", str(cert_pem),
        "-pubkey",
        "-noout",
    ])
    if result.returncode != 0:
        fail("Unable to extract public key from certificate.")

    tmp = Path(tempfile.mkstemp(prefix="cert_pub_", suffix=".pem")[1])
    try:
        tmp.write_text(result.stdout)
        der = run([
            "openssl", "pkey",
            "-pubin",
            "-in", str(tmp),
            "-outform", "DER",
        ])
        if der.returncode != 0:
            fail("Unable to normalize certificate public key.")
        return hashlib.sha256(der.stdout.encode() if isinstance(der.stdout, str)
                              else der.stdout).hexdigest()
    finally:
        tmp.unlink(missing_ok=True)


def public_key_hash_from_key(key_pem: Path) -> str:
    result = run([
        "openssl", "pkey",
        "-in", str(key_pem),
        "-pubout",
    ])
    if result.returncode != 0:
        fail("Unable to extract public key from private key.")

    tmp = Path(tempfile.mkstemp(prefix="key_pub_", suffix=".pem")[1])
    try:
        tmp.write_text(result.stdout)
        der = run([
            "openssl", "pkey",
            "-pubin",
            "-in", str(tmp),
            "-outform", "DER",
        ])
        if der.returncode != 0:
            fail("Unable to normalize private-key public key.")
        return hashlib.sha256(der.stdout.encode() if isinstance(der.stdout, str)
                              else der.stdout).hexdigest()
    finally:
        tmp.unlink(missing_ok=True)


def show_certificate_info(cert_pem: Path) -> None:
    result = run([
        "openssl", "x509",
        "-in", str(cert_pem),
        "-noout",
        "-subject",
        "-issuer",
        "-serial",
        "-dates",
    ])
    if result.returncode == 0:
        print("\n[+] Client certificate information:")
        print(result.stdout.strip())


def check_validity(cert_pem: Path) -> None:
    result = run([
        "openssl", "x509",
        "-in", str(cert_pem),
        "-checkend", "0",
        "-noout",
    ])
    if result.returncode == 0:
        print("[+] Certificate is currently valid according to OpenSSL.")
    else:
        print("[!] WARNING: Certificate is expired or not yet valid.")


def check_eku(cert_pem: Path) -> None:
    result = run([
        "openssl", "x509",
        "-in", str(cert_pem),
        "-noout",
        "-ext", "extendedKeyUsage",
    ])

    text = (result.stdout or "").strip()
    if result.returncode != 0 or not text:
        print("[!] Extended Key Usage extension not present/readable.")
        return

    print("\n[+] Extended Key Usage:")
    print(text)

    if re.search(
        r"TLS Web Client Authentication|clientAuth|Client Authentication",
        text,
        flags=re.IGNORECASE,
    ):
        print("[+] Client Authentication EKU detected.")
    else:
        print("[!] WARNING: Client Authentication EKU was not detected.")


def extract_pem_cert_chain(cert_path: Path, workdir: Path) -> Path | None:
    try:
        text = cert_path.read_text(errors="strict")
    except UnicodeDecodeError:
        return None

    blocks = re.findall(
        r"-----BEGIN CERTIFICATE-----.*?-----END CERTIFICATE-----",
        text,
        flags=re.DOTALL,
    )

    if len(blocks) <= 1:
        return None

    chain_path = workdir / "chain.pem"
    chain_path.write_text("\n".join(blocks[1:]) + "\n")
    print(f"[+] Detected {len(blocks)} certificate blocks in the PEM file.")
    print(f"[+] Including {len(blocks) - 1} additional certificate(s) as chain.")
    return chain_path


def create_pkcs12(
    cert_pem: Path,
    key_pem: Path,
    chain_pem: Path | None,
    output: Path,
    p12_password: str,
) -> None:
    cmd = [
        "openssl", "pkcs12",
        "-export",
        "-out", str(output),
        "-in", str(cert_pem),
        "-inkey", str(key_pem),
        "-name", "Burp-mTLS-client",
        "-passout", "stdin",
    ]

    if chain_pem:
        cmd += ["-certfile", str(chain_pem)]

    result = run(cmd, stdin_text=p12_password + "\n")
    if result.returncode != 0:
        fail("PKCS#12 creation failed:\n" + result.stderr.strip())


def verify_pkcs12(output: Path, p12_password: str) -> None:
    result = run([
        "openssl", "pkcs12",
        "-info",
        "-in", str(output),
        "-noout",
        "-passin", "stdin",
    ], stdin_text=p12_password + "\n")

    if result.returncode != 0:
        fail(
            "Generated client.p12 could not be opened with the password "
            "you entered:\n" + result.stderr.strip()
        )

    combined = ((result.stdout or "") + (result.stderr or "")).lower()

    # OpenSSL writes some PKCS#12 details to stderr.
    if "private key" not in combined:
        print("[!] WARNING: Could not clearly confirm a private key in the PKCS#12.")
    else:
        print("[+] Private key present in PKCS#12.")

    if "certificate" not in combined:
        print("[!] WARNING: Could not clearly confirm a certificate in the PKCS#12.")
    else:
        print("[+] Certificate present in PKCS#12.")
